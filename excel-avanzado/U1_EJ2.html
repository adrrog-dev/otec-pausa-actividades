Ejercicio 1 — Lectura de código: ¿qué hace este bucle?

Sub ResaltarNegativos()
    Dim i As Long
    For i = 2 To 100
        If Cells(i, 3).Value < 0 Then
            Cells(i, 3).Interior.Color = RGB(231, 76, 60)
        End If
    Next i
End Sub
